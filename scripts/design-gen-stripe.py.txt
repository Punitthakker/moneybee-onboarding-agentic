def prop(t):
    if isinstance(t,list):
        nul=any(x.startswith('?') for x in t); vals=[x.lstrip('?') for x in t]
        return ({'type':['string','null'],'enum':vals+[None]} if nul else {'type':'string','enum':vals}),False
    req=t.endswith('!'); t=t.rstrip('!')
    null=t.startswith('?'); t=t.lstrip('?')
    if t.startswith('['): s={'type':'array','items':ref_or(t[1:-1])}
    else: s=ref_or(t)
    if null:
        s={'oneOf':[{'type':'null'},s]} if '$ref' in s else {**s,'type':[s.get('type','string'),'null']}
    return s,req
def ref_or(t):
    if t in('string','integer','number','boolean','object'): return {'type':t}
    if t=='datetime': return {'type':'string','format':'date-time'}
    if t=='binary': return {'type':'string','format':'binary'}
    return {'$ref':f'#/components/schemas/{t}'}
def props(d):
    out={};req=[]
    for k,t in d.items():
        s_,r=prop(t); out[k]=s_
        if r: req.append(k)
    return out,req
def obj(d,extra=None,required=True):
    p,rq=props(d)
    o={'type':'object','properties':p,'additionalProperties':False}
    return o
